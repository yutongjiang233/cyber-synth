<script>

    // 纯洁的原生 JavaScript，无需任何外部依赖！

    let activeWave = 'Sine'; let activeModDest = 'Pitch'; let activeBpm = 120;

    function bindSegmentedControl(id, callback) { document.querySelectorAll(`#${id} .segment, #${id} .segment-blue`).forEach(seg => { seg.addEventListener('click', (e) => { document.querySelector(`#${id} .active`).classList.remove('active'); e.target.classList.add('active'); callback(e.target.getAttribute('data-wave') || e.target.getAttribute('data-mod') || e.target.getAttribute('data-bpm')); }); }); }

    bindSegmentedControl('waveform-control', val => activeWave = val); bindSegmentedControl('mod-control', val => activeModDest = val); bindSegmentedControl('bpm-control', val => activeBpm = val);


    const btn = document.getElementById("btn-generate"); const statusBox = document.getElementById("status-box"); const audio = document.getElementById("audio-player"); const timbreText = document.getElementById("timbre-text");


    function constructPrompt() {

        const attack = parseInt(document.getElementById("attack").value); const release = parseInt(document.getElementById("release").value); const lfoDepth = parseInt(document.getElementById("lfo-depth").value); const timbre = timbreText.value || "synthesized textures";

        let prompt = `electronic music, ${activeBpm} BPM, `;

        if (activeWave === 'Sine') prompt += "smooth sine wave tones, "; else if (activeWave === 'Sawtooth') prompt += "bright sawtooth, "; else prompt += "complex wave, ";

        if (attack > 70) prompt += "slow evolving atmospheric pads, "; else if (attack < 30) prompt += "sharp staccato, ";

        if (release > 70) prompt += "long infinite reverb tail, ";

        if (lfoDepth > 20) prompt += `LFO modulation on ${activeModDest}, `;

        return prompt + timbre;

    }


    btn.addEventListener("click", async () => {

        const fullPrompt = constructPrompt();

        btn.disabled = true; btn.innerText = "⏳ Requesting Cloud GPU..."; audio.src = "";

        statusBox.innerHTML = `<span class="loading">Sending parameters to Vercel Gateway...</span><br>Prompt:<br><span id="final-prompt">'${fullPrompt}'</span>`;


        try {

            // 1. 发送请求给 Vercel 后端

            const response = await fetch("/api/predict", {

                method: "POST",

                headers: { "Content-Type": "application/json" },

                body: JSON.stringify({ prompt: fullPrompt })

            });

            let prediction = await response.json();

            if (response.status !== 201) throw new Error(prediction.detail || "Gateway Error");


            // 2. 轮询循环：每 2 秒问一次后端做好了没

            while (prediction.status !== "succeeded" && prediction.status !== "failed") {

                await new Promise(r => setTimeout(r, 2000));

                const pollRes = await fetch(`/api/predict?id=${prediction.id}`);

                prediction = await pollRes.json();

                statusBox.innerHTML = `<span class="loading">GPU Cluster Status: [${prediction.status.toUpperCase()}]...</span><br>Prompt:<br><span id="final-prompt">'${fullPrompt}'</span>`;

            }


            // 3. 拿到音频！

            if (prediction.status === "succeeded") {

                audio.src = prediction.output; // Replicate 直接返回音频链接

                statusBox.innerHTML = `Ritual Complete.<br>Audio fully manifested from latent space.`;

            } else {

                throw new Error("Generation failed on Replicate.");

            }

        } catch (error) {

            statusBox.innerHTML = `❌ RITUAL ERROR: ${error.message}`;

        } finally {

            btn.disabled = false; btn.innerText = "🔮 SYNTHESIZE AUDIO RITUAL";

        }

    });

</script>
